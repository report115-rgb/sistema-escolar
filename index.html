class AsaasIntegrationService { constructor() { const config = JSON.parse(localStorage.getItem('app_asaas_config')) || { apiKey: '', env: 'sandbox' }; this.apiKey = config.apiKey ? config.apiKey.trim() : ''; this.environment = config.env || 'sandbox'; this.rawBaseURL = this.environment === 'production' ? 'https://api.asaas.com/v3' : 'https://sandbox.asaas.com/v3'; } getURL(endpoint) { const target = `${this.rawBaseURL}${endpoint}`; return `https://corsproxy.io/?${encodeURIComponent(target)}`; } getHeaders() { return { 'Content-Type': 'application/json', 'access_token': this.apiKey }; } async criarOuObterCliente(aluno) { if (!this.apiKey) throw new Error("Chave de API do Asaas não configurada."); const cpfCnpjLimpo = aluno.cpf ? aluno.cpf.replace(/\D/g, '') : ''; try { // 1. Busca cliente pelo CPF const searchRes = await fetch(this.getURL(`/customers?cpfCnpj=${cpfCnpjLimpo}`), { method: 'GET', headers: this.getHeaders() }); const searchData = await searchRes.json(); if (searchData.data && searchData.data.length > 0) { return searchData.data[0].id; } // 2. Se não existir, cadastra o cliente enviando um objeto limpo const payloadCliente = { name: aluno.fullname, email: aluno.email || "aluno@se7educacional.com", cpfCnpj: cpfCnpjLimpo, phone: aluno.whatsapp ? aluno.whatsapp.replace(/\D/g, '') : '91999999999' }; const createRes = await fetch(this.getURL('/customers'), { method: 'POST', headers: this.getHeaders(), body: JSON.stringify(payloadCliente) }); const responseText = await createRes.text(); let createData; try { createData = JSON.parse(responseText); } catch (err) { console.error("Retorno bruto não JSON do Asaas/Proxy:", responseText); throw new Error("O servidor intermediário bloqueou a resposta do Asaas."); } if (!createRes.ok || createData.errors) { const detalheErro = createData.errors ? createData.errors.map(e => e.description).join(', ') : 'Erro desconhecido ao cadastrar cliente'; throw new Error(detalheErro); } return createData.id; } catch (err) { console.error("Erro na integração com Asaas (Cliente):", err); throw err; } } async criarCobrancaBoleto(customerId, dadosCobranca) { const payload = { customer: customerId, billingType: 'UNDEFINED', value: parseFloat(dadosCobranca.valor), dueDate: dadosCobranca.vencimento, description: dadosCobranca.descricao || 'Taxa Educacional SE7', externalReference: dadosCobranca.referencia || `SE7_${Date.now()}` }; const res = await fetch(this.getURL('/payments'), { method: 'POST', headers: this.getHeaders(), body: JSON.stringify(payload) }); const responseText = await res.text(); let data; try { data = JSON.parse(responseText); } catch (err) { throw new Error("Falha ao interpretar resposta de cobrança do Asaas."); } if (!res.ok || data.errors) { const detalheErro = data.errors ? data.errors.map(e => e.description).join(', ') : 'Erro ao gerar cobrança'; throw new Error(detalheErro); } return { paymentId: data.id, invoiceUrl: data.invoiceUrl, bankSlipUrl: data.bankSlipUrl, identificationField: data.identificationField }; } }
