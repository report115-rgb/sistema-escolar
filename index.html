// ========================================== // FUNÇÕES DO CALENDÁRIO ACADÊMICO FLUTUANTE // ========================================== function abrirModalCalendarioFlutuante() { const modal = document.getElementById('modal-calendario-flutuante'); if (!modal) return; // Preenche o select de cursos com os cursos já cadastrados no sistema const selectCurso = document.getElementById('float-cal-course'); if (selectCurso) { selectCurso.innerHTML = ''; if (typeof courses !== 'undefined' && courses.length > 0) { courses.forEach(c => { selectCurso.insertAdjacentHTML('beforeend', ``); }); } } // Limpa as turmas e resultados ao abrir document.getElementById('float-cal-class').innerHTML = ''; document.getElementById('float-cal-result-container').innerHTML = `

Selecione o curso e a turma acima para visualizar o calendário acadêmico.

`; modal.style.display = 'flex'; } function fecharModalCalendarioFlutuante() { const modal = document.getElementById('modal-calendario-flutuante'); if (modal) modal.style.display = 'none'; } // Atalho global para fechar com a tecla ESC window.addEventListener('keydown', (e) => { if (e.key === 'Escape') { fecharModalCalendarioFlutuante(); } }); function atualizarTurmasCalendarioFlutuante() { const cursoId = document.getElementById('float-cal-course').value; const selectTurma = document.getElementById('float-cal-class'); if (!selectTurma) return; selectTurma.innerHTML = ''; if (!cursoId) { document.getElementById('float-cal-result-container').innerHTML = `

Selecione o curso e a turma acima para visualizar o calendário acadêmico.

`; return; } if (typeof classes !== 'undefined') { const turmasFiltradas = classes.filter(t => t.courseId === cursoId); if (turmasFiltradas.length === 0) { selectTurma.innerHTML = ''; } else { turmasFiltradas.forEach(t => { selectTurma.insertAdjacentHTML('beforeend', ``); }); } } renderizarCalendarioFlutuanteResultado(); } function renderizarCalendarioFlutuanteResultado() { const turmaId = document.getElementById('float-cal-class').value; const semestreSelecionado = document.getElementById('float-cal-semester').value; const container = document.getElementById('float-cal-result-container'); if (!container) return; if (!turmaId) { container.innerHTML = `

Selecione uma turma para carregar os eventos.

`; return; } if (typeof academicCalendars === 'undefined' || academicCalendars.length === 0) { container.innerHTML = `

Nenhum calendário acadêmico cadastrado no sistema.

`; return; } // Filtra os eventos da turma e do semestre escolhido const eventosTurma = academicCalendars.filter(ev => ev.classId === turmaId && (!ev.semestre || ev.semestre === semestreSelecionado) ); if (eventosTurma.length === 0) { container.innerHTML = `

Nenhuma aula ou evento programado para esta turma no semestre ${semestreSelecionado}.

`; return; } let htmlEventos = ''; // Organiza e exibe os eventos de forma limpa eventosTurma.forEach(ev => { const datasFormatadas = (ev.dates || []).map(d => d.split('-').reverse().join('/')).join(', '); htmlEventos += `
${ev.subject} ${ev.semestre || semestreSelecionado}

${ev.description || 'Aula programada'}

Datas de Aula: ${datasFormatadas || 'N/I'}
`; }); container.innerHTML = htmlEventos; }
