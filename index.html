function openSubTab(evt, subtabId) { // Localiza o container principal da aba Sistema const tabSettings = document.getElementById('tab-settings'); if (tabSettings) { // Esconde TODAS as sub-abas de sistema tabSettings.querySelectorAll('.subtab-content').forEach(el => { el.style.display = 'none'; }); // Remove o destaque de todos os botões de sub-aba tabSettings.querySelectorAll('.subtab-button').forEach(el => { el.classList.remove('bg-blue-600', 'text-white', 'shadow'); el.classList.add('bg-gray-100', 'dark:bg-slate-700', 'text-gray-700', 'dark:text-gray-300'); }); } // Exibe especificamente a sub-aba solicitada const targetSubTab = document.getElementById(subtabId); if (targetSubTab) { targetSubTab.style.display = 'block'; } // Destaca o botão correspondente if (evt && evt.currentTarget) { evt.currentTarget.classList.remove('bg-gray-100', 'dark:bg-slate-700', 'text-gray-700', 'dark:text-gray-300'); evt.currentTarget.classList.add('bg-blue-600', 'text-white', 'shadow'); } // Dispara as funções de renderização específicas de cada aba para garantir dados na tela if(subtabId === 'subtab-calendar') setTimeout(renderizarCalendarioSemestralAdmin, 200); if(subtabId === 'subtab-subjects') renderSubjectsManagement(); if(subtabId === 'subtab-classes') renderClasses(); }
