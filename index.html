async function addOrUpdateStudent() { const editId = document.getElementById('edit-student-id').value; const fullname = document.getElementById('nomeAluno').value.trim(); const email = document.getElementById('emailAluno').value.trim(); const cpf = document.getElementById('cpfAluno').value.trim(); const courseId = document.getElementById('input-student-course').value; const classId = document.getElementById('input-student-class').value; if (!fullname || !email || !cpf || !courseId || !classId) return alert("Preencha todos os campos!"); if (!validarCPF(cpf)) return alert("CPF inválido!"); const cursoObj = courses.find(c => c.id === courseId); const studentData = { fullname, dob: document.getElementById('dataNascimento').value, sexo: document.getElementById('sexoAluno').value, whatsapp: document.getElementById('whatsappAluno').value.trim(), email, rg: document.getElementById('rgAluno').value.trim(), cpf, cep: document.getElementById('cepAluno').value.trim(), endereco: document.getElementById('enderecoAluno').value.trim(), numero: document.getElementById('numeroAluno').value.trim(), bairro: document.getElementById('bairroAluno').value.trim(), cidade: document.getElementById('cidadeAluno').value.trim(), estado: document.getElementById('estadoAluno').value.trim(), courseId, classId }; if (editId) { await db.collection("students").doc(editId).update(studentData); closeNewStudentModal(); alert("Dados do aluno atualizados com sucesso!"); } else { const code = `MAT-${new Date().getFullYear()}-${Date.now().toString().slice(-5)}`; studentData.code = code; studentData.status = "Ativo"; studentData.senhaAluno = code; studentData.documentos = {}; studentData.fotoAluno = ""; const docRef = await db.collection("students").add(studentData); const novoAlunoId = docRef.id; closeNewStudentModal(); // --- GATILHO AUTOMÁTICO DE GERAÇÃO FINANCEIRA --- try { if (cursoObj && cursoObj.valorParcela && cursoObj.qtdParcelas) { const qtdP = cursoObj.qtdParcelas; const valP = cursoObj.valorParcela; const valDesc = cursoObj.valorDesconto || valP; const diaVenc = cursoObj.vencimento || 10; for (let i = 1; i <= qtdP; i++) { const dataV = new Date(); dataV.setMonth(dataV.getMonth() + (i - 1)); dataV.setDate(diaVenc); const dataStr = dataV.toISOString().slice(0, 10); await db.collection("financeLogs").add({ studentId: novoAlunoId, studentName: fullname, studentCode: code, description: `Mensalidade ${i}/${qtdP} - ${cursoObj.name}`, amount: parseFloat(valP), discountAmount: parseFloat(valDesc), date: dataStr, status: "PENDENTE (Desc. Pontualidade disponível até o vencimento)" }); } } else { await db.collection("financeLogs").add({ studentId: novoAlunoId, studentName: fullname, studentCode: code, description: `Matrícula / Taxa Inicial - ${cursoObj ? cursoObj.name : 'Curso Geral'}`, amount: 0.00, discountAmount: 0.00, date: new Date().toISOString().slice(0, 10), status: "PENDENTE" }); } } catch (erroFin) { console.error("Erro ao gerar financeiro automático:", erroFin); } notificarCentralWhatsAppMatriz(studentData, cursoObj ? cursoObj.name : "Curso Geral"); } }
