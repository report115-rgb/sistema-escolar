Salva o agendamento individual do aluno no Firebase async function salvarAgendamentoProvaAluno(studentId) { const dtInicio = document.getElementById(`dt_inicio_${studentId}`).value; const dtFim = document.getElementById(`dt_fim_${studentId}`).value; if (!dtInicio || !dtFim) { return alert("Preencha a data e hora de início e término."); } try { const aluno = students.find(s => s.id === studentId); if (!aluno.controleEJA) aluno.controleEJA = {}; aluno.controleEJA.dataInicio = dtInicio; aluno.controleEJA.dataFim = dtFim; await db.collection("students").doc(studentId).set({ controleEJA: aluno.controleEJA }, { merge: true }); alert("Agendamento da prova salvo com sucesso!
