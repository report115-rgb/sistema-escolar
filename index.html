async criarOuObterCliente(aluno) { if (!this.apiKey) throw new Error("Chave de API do Asaas não configurada."); const cpfCnpjLimpo = aluno.cpf ? aluno.cpf.replace(/\D/g, '') : ''; try { // Busca cliente pelo CPF usando o proxy const searchRes = await fetch(this.getURL(`/customers?cpfCnpj=${cpfCnpjLimpo}`), { method: 'GET', headers: this.getHeaders() }); const rawSearchText = await searchRes.text(); let searchData; try { searchData = JSON.parse(rawSearchText); } catch (e) { console.error("Resposta inválida do proxy/Asaas (Busca):", rawSearchText); throw new Error("O proxy CORS retornou uma página de erro ou bloqueio em vez de dados."); } if (searchData.data && searchData.data.length > 0) { return searchData.data[0].id; } // Se não existir, cadastra o cliente no Asaas const createRes = await fetch(this.getURL('/customers'), { method: 'POST', headers: this.getHeaders(), body: JSON.stringify({ name: aluno.fullname, email: aluno.email || "aluno@se7educacional.com", cpfCnpj: cpfCnpjLimpo, phone: aluno.whatsapp ? aluno.whatsapp.replace(/\D/g, '') : '91999999999' }) }); const rawCreateText = await createRes.text(); let createData; try { createData = JSON.parse(rawCreateText); } catch (e) { console.error("Resposta inválida do proxy/Asaas (Criação):", rawCreateText); throw new Error("O Asaas recusou a requisição ou o proxy bloqueou o envio dos dados."); } if (!createRes.ok) { const detalheErro = createData.errors ? createData.errors.map(e => e.description).join(', ') : 'Erro desconhecido do Asaas'; throw new Error(`Asaas recusou: ${detalheErro}`); } return createData.id; } catch (err) { console.error("Erro na integração com Asaas (Cliente):", err); throw err; } }
