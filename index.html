function openTab(evt, tabId) { // 1. Oculta TODAS as abas principais do sistema const tabContents = document.getElementsByClassName("tab-content"); for (let i = 0; i < tabContents.length; i++) { tabContents[i].style.display = "none"; } // 2. Remove o estilo ativo de todos os botões da barra de navegação principal const tabButtons = document.getElementsByClassName("tab-button"); for (let i = 0; i < tabButtons.length; i++) { // Preserva o botão de destaque do Portal do Aluno se necessário, ou reseta o estilo geral if(tabButtons[i].id !== 'nav-btn-portal-aluno') { tabButtons[i].classList.remove("bg-blue-600", "text-white", "shadow"); tabButtons[i].classList.add("text-gray-700", "dark:text-gray-200", "hover:bg-gray-300", "dark:hover:bg-slate-600"); } } // --- CORREÇÃO CRUCIAL AQUI: LIMPEZA DE SUBABAS --- // Oculta e reseta todas as subabas internas do sistema ao transitar entre abas principais const todasSubabas = document.querySelectorAll('.subtab-content'); todasSubabas.forEach(el => { el.style.display = "none"; el.classList.remove("active"); }); // Reseta o estado visual dos botões de subabas para o padrão inativo const todosBotoesSub = document.querySelectorAll('.subtab-button'); todosBotoesSub.forEach(el => { el.classList.remove("bg-blue-600", "text-white", "shadow"); el.classList.add("bg-gray-100", "dark:bg-slate-700", "text-gray-700", "dark:text-gray-300"); }); // ------------------------------------------------ // 3. Exibe estritamente a aba principal selecionada const abaAlvo = document.getElementById(tabId); if (abaAlvo) { abaAlvo.style.display = "block"; } // 4. Aplica o estilo ativo no botão principal clicado if (evt && evt.currentTarget) { if(evt.currentTarget.id !== 'nav-btn-portal-aluno') { evt.currentTarget.classList.remove("text-gray-700", "dark:text-gray-200", "hover:bg-gray-300", "dark:hover:bg-slate-600"); evt.currentTarget.classList.add("bg-blue-600", "text-white", "shadow"); } } // 5. Gatilhos de renderização específica para componentes que dependem do DOM visível if(tabId === 'tab-settings') setTimeout(renderizarCalendarioSemestralAdmin, 200); if(tabId === 'tab-portal-aluno') setTimeout(renderizarCalendarioSemestralPortalAluno, 200); if(tabId === 'tab-audit') setTimeout(renderizarLogsAuditoria, 200); }
